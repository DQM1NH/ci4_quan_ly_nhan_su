-- =========================================
-- 20. VIEW CHI TIET NHAN VIEN
-- =========================================

CREATE VIEW v_chi_tiet_nhan_vien AS
SELECT
    nv.nhan_vien_id,
    nv.ma_nhan_vien,
    nv.ho_ten,
    nv.gioi_tinh,
    nv.so_dien_thoai,
    nv.email,
    pb.ten_phong_ban,
    vt.ten_vai_tro,
    nv.luong,
    nv.trang_thai
FROM nhan_vien nv
LEFT JOIN phong_ban pb
ON nv.phong_ban_id = pb.phong_ban_id
LEFT JOIN vai_tro vt
ON nv.vai_tro_id = vt.vai_tro_id;

-- =========================================
-- 21. VIEW CHI TIET BANG LUONG
-- =========================================

CREATE VIEW v_chi_tiet_bang_luong AS
SELECT
    bl.bang_luong_id,
    nv.ho_ten,
    kl.thang,
    kl.nam,
    bl.luong_co_ban,
    bl.phu_cap,
    bl.thuong,
    bl.luong_tang_ca,
    bl.khau_tru,
    bl.bao_hiem,
    bl.thue,
    bl.luong_thuc_nhan,
    bl.trang_thai_thanh_toan
FROM bang_luong bl
JOIN nhan_vien nv
ON bl.nhan_vien_id = nv.nhan_vien_id
JOIN ky_luong kl
ON bl.ky_luong_id = kl.ky_luong_id;

-- =========================================
-- 22. STORED PROCEDURE TINH LUONG
-- =========================================

DELIMITER $$

CREATE PROCEDURE sp_tinh_luong(
    IN p_nhan_vien_id INT,
    IN p_ky_luong_id INT,
    IN p_luong_co_ban DECIMAL(15,2),
    IN p_phu_cap DECIMAL(15,2),
    IN p_thuong DECIMAL(15,2),
    IN p_luong_tang_ca DECIMAL(15,2),
    IN p_khau_tru DECIMAL(15,2),
    IN p_bao_hiem DECIMAL(15,2),
    IN p_thue DECIMAL(15,2)
)
BEGIN
    DECLARE v_luong_thuc_nhan DECIMAL(15,2);
    SET v_luong_thuc_nhan =
        p_luong_co_ban
        + p_phu_cap
        + p_thuong
        + p_luong_tang_ca
        - p_khau_tru
        - p_bao_hiem
        - p_thue;
    INSERT INTO bang_luong(
        nhan_vien_id,
        ky_luong_id,
        luong_co_ban,
        phu_cap,
        thuong,
        luong_tang_ca,
        khau_tru,
        bao_hiem,
        thue,
        luong_thuc_nhan
    )
    VALUES(
        p_nhan_vien_id,
        p_ky_luong_id,
        p_luong_co_ban,
        p_phu_cap,
        p_thuong,
        p_luong_tang_ca,
        p_khau_tru,
        p_bao_hiem,
        p_thue,
        v_luong_thuc_nhan
    );

END $$

DELIMITER ;

-- =========================================
-- 23. TRIGGER TU DONG TINH GIO LAM
-- =========================================

DELIMITER $$

CREATE TRIGGER trg_tinh_gio_lam
BEFORE UPDATE ON cham_cong
FOR EACH ROW
BEGIN

    IF NEW.check_in IS NOT NULL
       AND NEW.check_out IS NOT NULL THEN

        SET NEW.gio_lam =
        TIMESTAMPDIFF(
            MINUTE,
            NEW.check_in,
            NEW.check_out
        ) / 60;

    END IF;

END $$

DELIMITER ;
